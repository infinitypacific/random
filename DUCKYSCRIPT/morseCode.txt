REM TITLE Morse code writer
REM AUTHOR Frostbyte
REM DESCRIPTION Type morse code using your caps lock key!
DEFINE #POLL_INTERVAL 16
DEFINE #DASH_THRESHOLD 4
DEFINE #MAX_PASSCODE 2

VAR $TICKS = 0
VAR $RECORDING = FALSE
VAR $LAST_NUM = FALSE
VAR $LAST_CODE = 0
VAR $RUNNING = TRUE
VAR $LAST_CAPS = FALSE
VAR $PASSCODE_ATTEMPTS = 0

EXTENSION PASSCODE_ENTRY_CAPSONLY
    REM VERSION 1.1
    REM AUTHOR: Frostbyte
    DEFINE #PASSCODE_RESPONSE_DELAY 20
    DEFINE #PASSCODE_INITIAL_ITERATION_LIMIT 1000
    DEFINE #PASSCODE_WINDOW 1

    VAR $CC_WAIT = 5
    FUNCTION CAPS_CHECK_LEEWAY()
        IF ($_CAPSLOCK_ON == TRUE) THEN
            VAR $CC_IS_ON = FALSE
        ELSE
            VAR $CC_IS_ON = TRUE
        END_IF

        VAR $C = 0
        WHILE (($_CAPSLOCK_ON != $CC_IS_ON) && ($C < $CC_WAIT))
            DELAY #PASSCODE_RESPONSE_DELAY
            $C = ($C + 1)
        END_WHILE

        IF($C < $CC_WAIT) THEN
            RETURN FALSE
        END_IF

        WHILE (($_CAPSLOCK_ON != $CC_IS_ON) && ($C < ($CC_WAIT + #PASSCODE_WINDOW)))
            DELAY #PASSCODE_RESPONSE_DELAY
            $C = ($C + 1)
        END_WHILE

        IF($_CAPSLOCK_ON == $CC_IS_ON) THEN
            RETURN TRUE
        ELSE
            RETURN FALSE
        END_IF
    END_FUNCTION
END_EXTENSION

ATTACKMODE HID
EXTENSION DETECT_READY
    REM VERSION 1.1
    REM AUTHOR: Korben

    REM_BLOCK DOCUMENTATION
        USAGE:
            Extension runs inline (here)
            Place at beginning of payload (besides ATTACKMODE) to act as dynamic
            boot delay

        TARGETS:
            Any system that reflects CAPSLOCK will detect minimum required delay
            Any system that does not reflect CAPSLOCK will hit the max delay of 3000ms
    END_REM

    REM CONFIGURATION:
    DEFINE #RESPONSE_DELAY 25
    DEFINE #ITERATION_LIMIT 120

    VAR $C = 0
    WHILE (($_CAPSLOCK_ON == FALSE) && ($C < #ITERATION_LIMIT))
        CAPSLOCK
        DELAY #RESPONSE_DELAY
        $C = ($C + 1)
    END_WHILE
    CAPSLOCK
END_EXTENSION

WHILE ( $RUNNING == TRUE )
    IF ($RECORDING == FALSE) THEN
        IF ($_SCROLLLOCK_ON == TRUE) THEN
            $RECORDING = TRUE

            $LAST_NUM = $_NUMLOCK_ON
            $TICKS = 0

            $LAST_CODE = 3
            EXFIL $LAST_CODE
        END_IF
        
        IF(($_CAPSLOCK_ON == TRUE) && ($PASSCODE_ATTEMPTS < #MAX_PASSCODE)) THEN
            $PASSCODE_ATTEMPTS = $PASSCODE_ATTEMPTS + 1
            $CC_WAIT = 3
            IF (CAPS_CHECK_LEEWAY() == TRUE) THEN
            IF (CAPS_CHECK_LEEWAY() == TRUE) THEN
            $CC_WAIT = 2
            IF (CAPS_CHECK_LEEWAY() == TRUE) THEN
            IF (CAPS_CHECK_LEEWAY() == TRUE) THEN
            IF (CAPS_CHECK_LEEWAY() == TRUE) THEN
            $RUNNING = FALSE
            END_IF
            END_IF
            END_IF
            END_IF
            END_IF
        END_IF
    ELSE
        REM Scroll Lock off: stop recording
        IF ($_SCROLLLOCK_ON == FALSE) THEN
            $RECORDING = FALSE
        END_IF

        IF ($RECORDING == TRUE) THEN

            REM Num Lock change: character separator
            IF ($_NUMLOCK_ON != $LAST_NUM) THEN
                STRING /
                $LAST_CODE = 2
                EXFIL $LAST_CODE
                $LAST_NUM = $_NUMLOCK_ON
                $TICKS = 0
            END_IF

            REM Caps Lock up: symbol
            IF ($_CAPSLOCK_ON != $LAST_CAPS) THEN
                IF ($_CAPSLOCK_ON == TRUE) THEN
                    $TICKS = 0
                ELSE
                    IF ($TICKS >= #DASH_THRESHOLD) THEN
                        STRING -
                        $LAST_CODE = 1
                        EXFIL $LAST_CODE
                    ELSE
                        STRING .
                        $LAST_CODE = 0
                        EXFIL $LAST_CODE
                    END_IF
                END_IF
                $LAST_CAPS = $_CAPSLOCK_ON
            END_IF

            $TICKS = ($TICKS + 1)
        END_IF
    END_IF

    DELAY #POLL_INTERVAL
END_WHILE
ATTACKMODE STORAGE
